"""Milvus 2.3 geo compatibility: keep vectors/IDs in Milvus, payloads in the client."""

from copy import copy

from engine.clients.milvus.eigen_configure_base import MilvusConfigurator as _Base


class MilvusConfigurator(_Base):
    def recreate(self, dataset, collection_params):
        schema = dataset.config.schema
        if not schema or set(schema.values()) != {"geo"}:
            raise ValueError("Milvus geo compatibility requires only geo payload fields")
        local = copy(dataset)
        local.config = copy(dataset.config)
        local.config.schema = {}
        # Original uploader sees only id/vector columns and keeps its selected ANN index.
        return super().recreate(local, collection_params)
